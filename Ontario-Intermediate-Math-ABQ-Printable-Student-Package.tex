\section{Quick Formula and Reference Boxes}

\begin{refbox}{Number and Algebra}
\begin{itemize}
  \item Exponent laws: $a^m\cdot a^n=a^{m+n}$, $\dfrac{a^m}{a^n}=a^{m-n}$ ($a\neq 0$), $(a^m)^n=a^{mn}$, $a^0=1$.
  \item Scientific notation: $a\times10^n$ where $1\le a<10$.
  \item Percent change: $\dfrac{\text{new}-\text{original}}{\text{original}}\times 100\%$.
  \item Linear equation form: $y=mx+b$ (slope $m$, $y$-intercept $b$).
\end{itemize}
\end{refbox}

\begin{refbox}{Data, Geometry, and Financial Literacy}
\begin{itemize}
  \item Mean: $\dfrac{\text{sum of values}}{\text{number of values}}$; median: middle value.
  \item Probability: $P(E)=\dfrac{\text{favourable outcomes}}{\text{total outcomes}}$.
  \item Distance formula: $d=\sqrt{(x_2-x_1)^2+(y_2-y_1)^2}$.
  \item Midpoint formula: $M\left(\dfrac{x_1+x_2}{2},\dfrac{y_1+y_2}{2}\right)$.
  \item Surface area of cylinder: $SA=2\pi r^2+2\pi rh$.
  \item Simple interest: $I=Prt$; amount $A=P+I$.
  \item Compound growth/decay: $A=P(1\pm r)^t$.
\end{itemize}
\end{refbox}

\begin{refbox}{Quadratics, Similarity, and Trigonometry}
\begin{itemize}
  \item Standard form: $y=ax^2+bx+c$; vertex form: $y=a(x-h)^2+k$.
  \item Axis of symmetry: $x=-\dfrac{b}{2a}$.
  \item Quadratic formula: $x=\dfrac{-b\pm\sqrt{b^2-4ac}}{2a}$.
  \item Similar triangles: corresponding side ratios are equal.
  \item SOH-CAH-TOA in right triangles.
  \item Sine law: $\dfrac{a}{\sin A}=\dfrac{b}{\sin B}=\dfrac{c}{\sin C}$.
  \item Cosine law: $c^2=a^2+b^2-2ab\cos C$.
\end{itemize}
\degmode
\end{refbox}

\section{Worked Examples (Review)}

\textbf{Example 1 (Exponents):} Simplify $\dfrac{(2^5)(2^{-3})}{2^2}$.\\
$\dfrac{2^{5-3}}{2^2}=\dfrac{2^2}{2^2}=2^0=1$.

\textbf{Example 2 (Linear equation):} Solve $3(2x+1)=21$.\\
$6x+3=21\Rightarrow 6x=18\Rightarrow x=3$.

\textbf{Example 3 (Quadratic factoring):} Solve $x^2-7x+12=0$.\\
$(x-3)(x-4)=0\Rightarrow x=3$ or $x=4$.

\textbf{Example 4 (Distance):} Distance between $(2,1)$ and $(8,10)$ is\\
$\sqrt{(8-2)^2+(10-1)^2}=\sqrt{36+81}=\sqrt{117}=3\sqrt{13}\approx10.82$.

\textbf{Example 5 (Trigonometry):} In a right triangle, opposite $=7$, adjacent $=24$. Find $\theta$.\\
$\tan\theta=\dfrac{7}{24}\Rightarrow \theta=\tan^{-1}(7/24)\approx 16.3^\circ$.\\
\degmode

\section{2--3 Day Study Schedule}
\subsection*{Two-Day Intensive Option}
\begin{itemize}
  \item \textbf{Day 1 (9--10 h):} Diagnostic, Set A, Set B, core quadratics, error-log review.
  \item \textbf{Day 2 (8--10 h):} Set C, mixed practice, mock assessment, corrections, readiness checklist.

\subsection*{Three-Day Option}
\begin{itemize}
  \item \textbf{Day 1:} Diagnostic + number/algebra + relations.

  \item \textbf{Day 3:} quadratics + analytic geometry + trigonometry + mock + final review.

\section{Final Readiness Checklist}
\begin{itemize}
  \item I can simplify expressions and solve linear equations accurately.
  \item I can identify and represent linear vs non-linear relations.
  \item I can compute midpoint, distance, and equations of lines.
  \item I can solve and interpret basic quadratic relations.
  \item I can use probability, data summaries, and bias language correctly.
  \item I can solve geometry and measurement problems with correct units.
  \item I can solve financial literacy problems (discounts, appreciation/depreciation, interest).
  \item I remember to use \textbf{degree mode} for trigonometry.
  \item I have reviewed my error log and corrected recurring mistakes.
  \item I can complete mixed and mock problems without heavy prompting.
\end{itemize}
